// ── TRACKING ──
function trackOrder() {
  const code = document.getElementById('track-input').value.trim().toUpperCase();
  const order = ordersDB[code];
  if(!order) {
    showToast('Order not found. Please check your tracking code.');
    document.getElementById('tracking-result').classList.remove('show');
    return;
  }
  const result = document.getElementById('tracking-result');
  result.classList.add('show');
  document.getElementById('tr-id').textContent = 'Order ' + order.id;
  document.getElementById('tr-name').textContent = order.items.map(i=>i.name+' ×'+i.qty).join(', ');
  const curStatus = statuses[order.statusIdx];
  const badge = document.getElementById('tr-status-badge');
  badge.textContent = curStatus;
  badge.className = 'track-current-status status-badge ' + statusColors[curStatus];
  // Timeline
  const tl = document.getElementById('tr-timeline');
  tl.innerHTML = '';
  statuses.forEach((s, i) => {
    const done = i < order.statusIdx;
    const active = i === order.statusIdx;
    const pending = i > order.statusIdx;
    const now = new Date();
    let timeStr = '';
    if(done) { const d=new Date(now - (order.statusIdx - i)*3600000*6); timeStr = d.toLocaleDateString('en-IN',{day:'numeric',month:'short'}) + ' · ' + d.toLocaleTimeString('en-IN',{hour:'2-digit',minute:'2-digit'}); }
    else if(active) { timeStr = 'Now'; }
    else { timeStr = 'Pending'; }
    const div = document.createElement('div');
    div.className = 'timeline-step' + (done?' done':'') + (active?' active':'');
    div.innerHTML = `<div class="timeline-dot">${statusEmojis[i]}</div><div class="timeline-info"><div class="timeline-label">${s}</div><div class="timeline-time">${timeStr}</div>${active?'<div class="timeline-desc">'+statusDescs[i]+'</div>':''}</div>`;
    tl.appendChild(div);
  });
  // Items
  const itemsDiv = document.getElementById('tr-items');
  itemsDiv.innerHTML = order.items.map(i=>`<div class="track-item-row"><span>${i.name} ×${i.qty}</span><span>₹${i.price*i.qty}</span></div>`).join('');
  document.getElementById('tr-total').textContent = order.total;
  result.scrollIntoView({behavior:'smooth', block:'start'});
}

function renderAdminOrders() {

  });

  const t=document.createElement('div');
  t.style.cssText='position:fixed;bottom:2rem;left:50%;transform:translateX(-50%);background:#1A1A18;color:white;padding:1rem 2rem;border-radius:100px;font-size:0.9rem;font-family:"DM Sans",sans-serif;z-index:9999;box-shadow:0 8px 30px rgba(0,0,0,0.3);animation:fadeInUp 0.4s ease;white-space:nowrap;';
  t.textContent=msg;
  document.body.appendChild(t);
  setTimeout(()=>t.remove(), 3500);
}

// ── AGNI SPARKS ──
function createSparks() {
  const container=document.getElementById('agni-sparks');
  if(!container) return;
  const colors=['#E53935','#FF6D00','#FFCA28','#FF8F00'];
  for(let i=0;i<40;i++){
    const s=document.createElement('div');
    s.className='spark';
    s.style.cssText=`left:${Math.random()*100}%;top:${Math.random()*100}%;background:${colors[Math.floor(Math.random()*colors.length)]};animation-delay:${Math.random()*3}s;animation-duration:${2+Math.random()*2}s;width:${2+Math.random()*4}px;height:${2+Math.random()*4}px;`;
    container.appendChild(s);
  }
}
createSparks();

  if(e.target.placeholder && e.target.placeholder.includes('1234 5678')) {
    let v=e.target.value.replace(/\D/g,'').substring(0,16);
    e.target.value=v.replace(/(.{4})/g,'$1 ').trim();
  }
});
